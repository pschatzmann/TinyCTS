100 bool connect(BLEDevice& peripheral) {
101 BLE.scan();
102 uint32_t start = millis();
103 bool found = false;
104 while (!found && static_cast<uint32_t>(millis() - start) < _scanMs) {
105 BLEDevice d = BLE.available();
106 if (!d) continue;
107 std::string address = d.address().c_str();
108 std::string name = d.hasLocalName() ? d.localName().c_str() : "";
109 if (_device.empty() ? offersService(d) : isWanted(address, name)) {
110 peripheral = d;
111 _address = address;
112 _name = name;
113 found = true;
114 }
115 }
116 BLE.stopScan();
117 if (!found || !peripheral.connect()) return false;
118 if (!peripheral.discoverAttributes()) {

120 return false;
121 }

125 static bool read(BLEDevice& peripheral, const char* uuid, uint8_t* out, size_t& len) {
126 BLECharacteristic chr = peripheral.characteristic(uuid);
127 if (!chr || !chr.canRead() || !chr.read()) return false;
128 len = static_cast<size_t>(chr.valueLength());
129 if (len == 0) return false;
130 if (len > 16) len = 16;
131 memcpy(out, chr.value(), len);
132 return true;
133 }

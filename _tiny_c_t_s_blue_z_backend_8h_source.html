19 bool scan(uint32_t timeoutMs, std::vector<TinyCTSScanEntry>& result,
20 std::string& error) override {
21 if (!open(error)) return false;
22 std::vector<TinyCTSBlueZ::Device> devices = TinyCTSBlueZ::discover(
23 _adapter, timeoutMs, [](const std::vector<TinyCTSBlueZ::Device>&) { return false; }, error);
24 for (const TinyCTSBlueZ::Device& d : devices) {
25 if (d.seen) result.push_back(entry(d));
26 }
27 return true;
28 }
29
30 bool connect(const std::string& wanted, uint32_t timeoutMs, TinyCTSScanEntry& found,
31 std::string& error) override {
32 disconnect();
33 if (!open(error)) return false;
34 std::vector<TinyCTSBlueZ::Device> devices = TinyCTSBlueZ::discover(
35 _adapter, timeoutMs,
36 [&](const std::vector<TinyCTSBlueZ::Device>& l) { return pick(l, wanted) != nullptr; }, error);
37 const TinyCTSBlueZ::Device* d = pick(devices, wanted);
38 if (d == nullptr) {
39 error = wanted.empty() ? "no device with the Current Time Service found"
40 : "device " + wanted + " not found";
41 return false;
42 }
43 _device = *d;
44 if (!_device.connected &&
45 TinyCTSBlueZ::call(_device.path, "org.bluez.Device1", "Connect", nullptr, error, 30000) ==
46 nullptr) {
47 error = "cannot connect to " + _device.address + ": " + error;
48 _device = TinyCTSBlueZ::Device();
49 return false;
50 }
51 for (int i = 0; i < 100; i++) { // up to 10 s
52 for (const TinyCTSBlueZ::Device& l : TinyCTSBlueZ::listDevices(_adapter)) {
53 if (l.path == _device.path && l.resolved) {
54 found = entry(_device);
55 return true;
56 }
57 }
58 g_usleep(100 * 1000);
59 }
60 error = "no services from " + _device.address;
61 disconnect();
62 return false;
63 }
64
65 bool read(uint16_t uuid, std::vector<uint8_t>& value, std::string& error) override {
66 std::string path = TinyCTSBlueZ::findCharacteristic(_device.path, uuid);
67 if (path.empty()) {
68 error = "the device does not have the characteristic";
69 return false;
70 }
71 return TinyCTSBlueZ::readValue(path, value, error);
72 }
73
74 bool write(uint16_t uuid, const uint8_t* data, size_t len, std::string& error) override {
75 std::string path = TinyCTSBlueZ::findCharacteristic(_device.path, uuid);
76 if (path.empty()) {
77 error = "the device does not have the characteristic";
78 return false;
79 }
80 return TinyCTSBlueZ::writeValue(path, data, len, error);
81 }
82
83 void disconnect() override {
84 if (_device.path.empty()) return;
85 std::string ignored;
86 TinyCTSBlueZ::call(_device.path, "org.bluez.Device1", "Disconnect", nullptr, ignored);
87 _device = TinyCTSBlueZ::Device();
88 }

91 bool open(std::string& error) {
92 error.clear();
93 _adapter = TinyCTSBlueZ::findAdapter(_adapterName, error);
94 return !_adapter.empty();
95 }

97 static TinyCTSScanEntry entry(const TinyCTSBlueZ::Device& d) {
98 TinyCTSScanEntry e;
99 e.address = d.address;
100 e.name = d.name;
101 e.rssi = d.rssi;
102 e.hasCts = d.cts;
103 return e;
104 }
105
106 static const TinyCTSBlueZ::Device* pick(const std::vector<TinyCTSBlueZ::Device>& devices,
107 const std::string& wanted) {
108 for (const TinyCTSBlueZ::Device& d : devices) {
109 if (wanted.empty()) {
110 if (d.cts && d.seen) return &d;
111 } else if (TinyCTSBlueZ::equalsIgnoreCase(d.address, wanted) || d.name == wanted) {
112 if (d.seen || d.connected) return &d;
113 }
114 }
115 return nullptr;
116 }

127 bool start(const std::string& name, bool settable, bool encrypted,
128 TinyCTSPeripheralHandler* handler, std::string& error) override {
129 _handler = handler;
130 _name = name;
131 _settable = settable;
132 _encrypted = encrypted;
133 _error.clear();
134 _adapter = TinyCTSBlueZ::findAdapter(_adapterName, _error);
135 GDBusConnection* conn = _adapter.empty() ? nullptr : TinyCTSBlueZ::bus(_error);
136 if (conn == nullptr) {
137 error = _error;
138 return false;
139 }
140
141 _nodeInfo = g_dbus_node_info_new_for_xml(xml(), nullptr);
142 _chars[0] = {std::string(Service) + "/char0", CTS_CURRENT_TIME_UUID, true};
143 _chars[1] = {std::string(Service) + "/char1", CTS_LOCAL_TIME_INFO_UUID, false};
144 _chars[2] = {std::string(Service) + "/char2", CTS_REFERENCE_TIME_INFO_UUID, false};
145 if (!exportObject(App, "org.freedesktop.DBus.ObjectManager") ||
146 !exportObject(Service, "org.bluez.GattService1") ||
147 !exportObject(_chars[0].path.c_str(), "org.bluez.GattCharacteristic1") ||
148 !exportObject(_chars[1].path.c_str(), "org.bluez.GattCharacteristic1") ||
149 !exportObject(_chars[2].path.c_str(), "org.bluez.GattCharacteristic1") ||
150 !exportObject(Advertisement, "org.bluez.LEAdvertisement1")) {
151 error = _error;
152 stop();
153 return false;
154 }
155 _started = true;
156
157 // BlueZ calls back while it registers: the main loop has to run
158 _pending = 2;
159 g_dbus_connection_call(conn, "org.bluez", _adapter.c_str(), "org.bluez.GattManager1",
160 "RegisterApplication", g_variant_new_parsed("(%o, @a{sv} {})", App),
161 nullptr, G_DBUS_CALL_FLAGS_NONE, 20000, nullptr, onRegistered, this);
162 g_dbus_connection_call(conn, "org.bluez", _adapter.c_str(),
163 "org.bluez.LEAdvertisingManager1", "RegisterAdvertisement",
164 g_variant_new_parsed("(%o, @a{sv} {})", Advertisement), nullptr,
165 G_DBUS_CALL_FLAGS_NONE, 20000, nullptr, onRegistered, this);
166 for (int i = 0; i < 4000 && _pending > 0; i++) { // up to 20 s
167 process();
168 g_usleep(5 * 1000);
169 }
170 if (_pending > 0 && _error.empty()) _error = "BlueZ does not answer";
171 if (!_error.empty()) {
172 error = _error;
173 stop();
174 return false;
175 }
176 _registered = true;
177 return true;
178 }

192 void stop() override {
193 std::string ignored;
194 if (_registered) {
195 TinyCTSBlueZ::call(_adapter, "org.bluez.GattManager1", "UnregisterApplication",
196 g_variant_new("(o)", App), ignored, 2000);
197 TinyCTSBlueZ::call(_adapter, "org.bluez.LEAdvertisingManager1", "UnregisterAdvertisement",
198 g_variant_new("(o)", Advertisement), ignored, 2000);
199 _registered = false;
200 }
201 GDBusConnection* conn = TinyCTSBlueZ::bus(ignored);
202 for (guint id : _ids) {
203 if (conn != nullptr) g_dbus_connection_unregister_object(conn, id);
204 }
205 _ids.clear();
206 if (_nodeInfo != nullptr) g_dbus_node_info_unref(_nodeInfo);
207 _nodeInfo = nullptr;
208 _started = false;
209 _notifying = false;
210 }

259 bool exportObject(const char* path, const char* iface) {
260 static const GDBusInterfaceVTable vtable = {onCall, onGet, nullptr, {}};
261 GError* err = nullptr;
262 GDBusConnection* conn = TinyCTSBlueZ::bus(_error);
263 GDBusInterfaceInfo* info = g_dbus_node_info_lookup_interface(_nodeInfo, iface);
264 guint id = conn == nullptr ? 0
265 : g_dbus_connection_register_object(conn, path, info, &vtable,
266 this, nullptr, &err);
267 if (id == 0) {
268 if (err != nullptr) {
269 _error = std::string("cannot export ") + path + ": " + err->message;
270 g_error_free(err);
271 }
272 return false;
273 }
274 _ids.push_back(id);
275 return true;
276 }

278 static void onRegistered(GObject* source, GAsyncResult* res, gpointer self) {
279 TinyCTSBlueZPeripheral* s = static_cast<TinyCTSBlueZPeripheral*>(self);
280 GError* err = nullptr;
281 GVariant* r = g_dbus_connection_call_finish(G_DBUS_CONNECTION(source), res, &err);
282 s->_pending--;

284 g_dbus_error_strip_remote_error(err);
285 s->_error = std::string("BlueZ refused the registration: ") + err->message;

349 static void onCall(GDBusConnection*, const gchar*, const gchar* path, const gchar*,
350 const gchar* method, GVariant* params, GDBusMethodInvocation* inv,
351 gpointer self) {
352 TinyCTSBlueZPeripheral* s = static_cast<TinyCTSBlueZPeripheral*>(self);
353 if (!strcmp(method, "GetManagedObjects")) {
354 g_dbus_method_invocation_return_value(inv, s->managed());
355 } else if (!strcmp(method, "ReadValue")) {
356 const Chr* c = s->find(path);
357 std::vector<uint8_t> value;
358 if (c == nullptr || s->_handler == nullptr || !s->_handler->onRead(c->uuid, value)) {
359 g_dbus_method_invocation_return_dbus_error(inv, "org.bluez.Error.Failed", "unknown");
360 return;
361 }
362 g_dbus_method_invocation_return_value(inv, g_variant_new("(@ay)", bytes(value)));
363 } else if (!strcmp(method, "WriteValue")) {
364 const Chr* c = s->find(path);
365 GVariant* data;
366 GVariant* options;
367 g_variant_get(params, "(@ay@a{sv})", &data, &options);
368 gsize n;
369 const uint8_t* d = static_cast<const uint8_t*>(g_variant_get_fixed_array(data, &n, 1));
370 bool ok = c != nullptr && s->_handler != nullptr && s->_handler->onWrite(c->uuid, d, n);
371 g_variant_unref(data);
372 g_variant_unref(options);
373 if (ok) {
374 g_dbus_method_invocation_return_value(inv, nullptr);
375 } else {
376 g_dbus_method_invocation_return_dbus_error(inv, "org.bluez.Error.NotPermitted",
377 "the time cannot be set");
378 }
379 } else if (!strcmp(method, "StartNotify")) {
380 s->_notifying = true;
381 g_dbus_method_invocation_return_value(inv, nullptr);
382 } else if (!strcmp(method, "StopNotify")) {
383 s->_notifying = false;
384 g_dbus_method_invocation_return_value(inv, nullptr);
385 } else { // Release
386 g_dbus_method_invocation_return_value(inv, nullptr);
387 }
388 }

390 static GVariant* onGet(GDBusConnection*, const gchar*, const gchar* path, const gchar* iface,
391 const gchar* prop, GError**, gpointer self) {
392 TinyCTSBlueZPeripheral* s = static_cast<TinyCTSBlueZPeripheral*>(self);
393 if (!strcmp(iface, "org.bluez.GattService1")) {
394 if (!strcmp(prop, "UUID")) {
395 return g_variant_new_string(TinyCTSBlueZ::fullUuid(CTS_SERVICE_UUID).c_str());
396 }
397 return g_variant_new_boolean(TRUE);
398 }
399 if (!strcmp(iface, "org.bluez.LEAdvertisement1")) {
400 if (!strcmp(prop, "Type")) return g_variant_new_string("peripheral");
401 if (!strcmp(prop, "LocalName")) return g_variant_new_string(s->_name.c_str());
402 std::string u = TinyCTSBlueZ::fullUuid(CTS_SERVICE_UUID);
403 const gchar* uuids[] = {u.c_str(), nullptr};
404 return g_variant_new_strv(uuids, 1);
405 }
406 const Chr* c = s->find(path);
407 if (c == nullptr) return nullptr;
408 if (!strcmp(prop, "UUID")) return g_variant_new_string(TinyCTSBlueZ::fullUuid(c->uuid).c_str());
409 if (!strcmp(prop, "Service")) return g_variant_new_object_path(Service);
410 if (!strcmp(prop, "Notifying")) return g_variant_new_boolean(s->_notifying);
411 return s->flags(*c);
412 }

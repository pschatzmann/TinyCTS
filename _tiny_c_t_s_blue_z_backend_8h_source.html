21 bool scan(uint32_t timeoutMs, std::vector<TinyCTSScanEntry>& result,
22 std::string& error) override {
23 if (!open(error)) return false;
24 std::vector<bluez::Device> devices = bluez::discover(
25 _adapter, timeoutMs, [](const std::vector<bluez::Device>&) { return false; }, error);
26 for (const bluez::Device& d : devices) {
27 if (d.seen) result.push_back(entry(d));
28 }
29 return true;
30 }
31
32 bool connect(const std::string& wanted, uint32_t timeoutMs, TinyCTSScanEntry& found,
33 std::string& error) override {
34 disconnect();
35 if (!open(error)) return false;
36 std::vector<bluez::Device> devices = bluez::discover(
37 _adapter, timeoutMs,
38 [&](const std::vector<bluez::Device>& l) { return pick(l, wanted) != nullptr; }, error);
39 const bluez::Device* d = pick(devices, wanted);
40 if (d == nullptr) {
41 error = wanted.empty() ? "no device with the Current Time Service found"
42 : "device " + wanted + " not found";
43 return false;
44 }
45 _device = *d;
46 if (!_device.connected &&
47 bluez::call(_device.path, "org.bluez.Device1", "Connect", nullptr, error, 30000) ==
48 nullptr) {
49 error = "cannot connect to " + _device.address + ": " + error;
50 _device = bluez::Device();
51 return false;
52 }
53 for (int i = 0; i < 100; i++) { // up to 10 s
54 for (const bluez::Device& l : bluez::listDevices(_adapter)) {
55 if (l.path == _device.path && l.resolved) {
56 found = entry(_device);
57 return true;
58 }
59 }
60 g_usleep(100 * 1000);
61 }
62 error = "no services from " + _device.address;
63 disconnect();
64 return false;
65 }
66
67 bool read(uint16_t uuid, std::vector<uint8_t>& value, std::string& error) override {
68 std::string path = bluez::findCharacteristic(_device.path, uuid);
69 if (path.empty()) {
70 error = "the device does not have the characteristic";
71 return false;
72 }
73 return bluez::readValue(path, value, error);
74 }
75
76 bool write(uint16_t uuid, const uint8_t* data, size_t len, std::string& error) override {
77 std::string path = bluez::findCharacteristic(_device.path, uuid);
78 if (path.empty()) {
79 error = "the device does not have the characteristic";
80 return false;
81 }
82 return bluez::writeValue(path, data, len, error);
83 }
84
85 void disconnect() override {
86 if (_device.path.empty()) return;
87 std::string ignored;
88 bluez::call(_device.path, "org.bluez.Device1", "Disconnect", nullptr, ignored);
89 _device = bluez::Device();
90 }

93 bool open(std::string& error) {
94 error.clear();
95 _adapter = bluez::findAdapter(_adapterName, error);
96 return !_adapter.empty();
97 }

99 static TinyCTSScanEntry entry(const bluez::Device& d) {
100 TinyCTSScanEntry e;
101 e.address = d.address;
102 e.name = d.name;
103 e.rssi = d.rssi;
104 e.hasCts = d.cts;
105 return e;
106 }

108 static const bluez::Device* pick(const std::vector<bluez::Device>& devices,
109 const std::string& wanted) {
110 for (const bluez::Device& d : devices) {
111 if (wanted.empty()) {
112 if (d.cts && d.seen) return &d;
113 } else if (bluez::equalsIgnoreCase(d.address, wanted) || d.name == wanted) {
114 if (d.seen || d.connected) return &d;
115 }
116 }
117 return nullptr;
118 }

129 bool start(const std::string& name, bool settable, bool encrypted,
130 PeripheralHandler* handler, std::string& error) override {
131 _handler = handler;
132 _name = name;
133 _settable = settable;
134 _encrypted = encrypted;
135 _error.clear();
136 _adapter = bluez::findAdapter(_adapterName, _error);
137 GDBusConnection* conn = _adapter.empty() ? nullptr : bluez::bus(_error);
138 if (conn == nullptr) {
139 error = _error;
140 return false;
141 }
142
143 _nodeInfo = g_dbus_node_info_new_for_xml(xml(), nullptr);
144 _chars[0] = {std::string(Service) + "/char0", CTS_CURRENT_TIME_UUID, true};
145 _chars[1] = {std::string(Service) + "/char1", CTS_LOCAL_TIME_INFO_UUID, false};
146 _chars[2] = {std::string(Service) + "/char2", CTS_REFERENCE_TIME_INFO_UUID, false};
147 if (!exportObject(App, "org.freedesktop.DBus.ObjectManager") ||
148 !exportObject(Service, "org.bluez.GattService1") ||
149 !exportObject(_chars[0].path.c_str(), "org.bluez.GattCharacteristic1") ||
150 !exportObject(_chars[1].path.c_str(), "org.bluez.GattCharacteristic1") ||
151 !exportObject(_chars[2].path.c_str(), "org.bluez.GattCharacteristic1") ||
152 !exportObject(Advertisement, "org.bluez.LEAdvertisement1")) {
153 error = _error;
154 stop();
155 return false;
156 }
157 _started = true;
158
159 // BlueZ calls back while it registers: the main loop has to run
160 _pending = 2;
161 g_dbus_connection_call(conn, "org.bluez", _adapter.c_str(), "org.bluez.GattManager1",
162 "RegisterApplication", g_variant_new_parsed("(%o, @a{sv} {})", App),
163 nullptr, G_DBUS_CALL_FLAGS_NONE, 20000, nullptr, onRegistered, this);
164 g_dbus_connection_call(conn, "org.bluez", _adapter.c_str(),
165 "org.bluez.LEAdvertisingManager1", "RegisterAdvertisement",
166 g_variant_new_parsed("(%o, @a{sv} {})", Advertisement), nullptr,
167 G_DBUS_CALL_FLAGS_NONE, 20000, nullptr, onRegistered, this);
168 for (int i = 0; i < 4000 && _pending > 0; i++) { // up to 20 s
169 process();
170 g_usleep(5 * 1000);
171 }
172 if (_pending > 0 && _error.empty()) _error = "BlueZ does not answer";
173 if (!_error.empty()) {
174 error = _error;
175 stop();
176 return false;
177 }
178 _registered = true;
179 return true;
180 }

194 void stop() override {
195 std::string ignored;
196 if (_registered) {
197 bluez::call(_adapter, "org.bluez.GattManager1", "UnregisterApplication",
198 g_variant_new("(o)", App), ignored, 2000);
199 bluez::call(_adapter, "org.bluez.LEAdvertisingManager1", "UnregisterAdvertisement",
200 g_variant_new("(o)", Advertisement), ignored, 2000);
201 _registered = false;
202 }
203 GDBusConnection* conn = bluez::bus(ignored);
204 for (guint id : _ids) {
205 if (conn != nullptr) g_dbus_connection_unregister_object(conn, id);
206 }
207 _ids.clear();
208 if (_nodeInfo != nullptr) g_dbus_node_info_unref(_nodeInfo);
209 _nodeInfo = nullptr;
210 _started = false;
211 _notifying = false;
212 }

261 bool exportObject(const char* path, const char* iface) {
262 static const GDBusInterfaceVTable vtable = {onCall, onGet, nullptr, {}};
263 GError* err = nullptr;
264 GDBusConnection* conn = bluez::bus(_error);
265 GDBusInterfaceInfo* info = g_dbus_node_info_lookup_interface(_nodeInfo, iface);
266 guint id = conn == nullptr ? 0
267 : g_dbus_connection_register_object(conn, path, info, &vtable,
268 this, nullptr, &err);
269 if (id == 0) {
270 if (err != nullptr) {
271 _error = std::string("cannot export ") + path + ": " + err->message;
272 g_error_free(err);
273 }
274 return false;
275 }
276 _ids.push_back(id);
277 return true;
278 }

280 static void onRegistered(GObject* source, GAsyncResult* res, gpointer self) {
281 BlueZPeripheral* s = static_cast<BlueZPeripheral*>(self);

283 GVariant* r = g_dbus_connection_call_finish(G_DBUS_CONNECTION(source), res, &err);

351 static void onCall(GDBusConnection*, const gchar*, const gchar* path, const gchar*,
352 const gchar* method, GVariant* params, GDBusMethodInvocation* inv,
353 gpointer self) {
354 BlueZPeripheral* s = static_cast<BlueZPeripheral*>(self);
355 if (!strcmp(method, "GetManagedObjects")) {
356 g_dbus_method_invocation_return_value(inv, s->managed());
357 } else if (!strcmp(method, "ReadValue")) {
358 const Chr* c = s->find(path);
359 std::vector<uint8_t> value;
360 if (c == nullptr || s->_handler == nullptr || !s->_handler->onRead(c->uuid, value)) {
361 g_dbus_method_invocation_return_dbus_error(inv, "org.bluez.Error.Failed", "unknown");
362 return;
363 }
364 g_dbus_method_invocation_return_value(inv, g_variant_new("(@ay)", bytes(value)));
365 } else if (!strcmp(method, "WriteValue")) {
366 const Chr* c = s->find(path);
367 GVariant* data;
368 GVariant* options;
369 g_variant_get(params, "(@ay@a{sv})", &data, &options);
370 gsize n;
371 const uint8_t* d = static_cast<const uint8_t*>(g_variant_get_fixed_array(data, &n, 1));
372 bool ok = c != nullptr && s->_handler != nullptr && s->_handler->onWrite(c->uuid, d, n);
373 g_variant_unref(data);
374 g_variant_unref(options);
375 if (ok) {
376 g_dbus_method_invocation_return_value(inv, nullptr);
377 } else {
378 g_dbus_method_invocation_return_dbus_error(inv, "org.bluez.Error.NotPermitted",
379 "the time cannot be set");
380 }
381 } else if (!strcmp(method, "StartNotify")) {
382 s->_notifying = true;
383 g_dbus_method_invocation_return_value(inv, nullptr);
384 } else if (!strcmp(method, "StopNotify")) {
385 s->_notifying = false;
386 g_dbus_method_invocation_return_value(inv, nullptr);
387 } else { // Release
388 g_dbus_method_invocation_return_value(inv, nullptr);
389 }
390 }

392 static GVariant* onGet(GDBusConnection*, const gchar*, const gchar* path, const gchar* iface,
393 const gchar* prop, GError**, gpointer self) {
394 BlueZPeripheral* s = static_cast<BlueZPeripheral*>(self);
395 if (!strcmp(iface, "org.bluez.GattService1")) {
396 if (!strcmp(prop, "UUID")) {
397 return g_variant_new_string(bluez::fullUuid(CTS_SERVICE_UUID).c_str());
398 }
399 return g_variant_new_boolean(TRUE);
400 }
401 if (!strcmp(iface, "org.bluez.LEAdvertisement1")) {
402 if (!strcmp(prop, "Type")) return g_variant_new_string("peripheral");
403 if (!strcmp(prop, "LocalName")) return g_variant_new_string(s->_name.c_str());
404 std::string u = bluez::fullUuid(CTS_SERVICE_UUID);
405 const gchar* uuids[] = {u.c_str(), nullptr};
406 return g_variant_new_strv(uuids, 1);
407 }
408 const Chr* c = s->find(path);
409 if (c == nullptr) return nullptr;
410 if (!strcmp(prop, "UUID")) return g_variant_new_string(bluez::fullUuid(c->uuid).c_str());
411 if (!strcmp(prop, "Service")) return g_variant_new_object_path(Service);
412 if (!strcmp(prop, "Notifying")) return g_variant_new_boolean(s->_notifying);
413 return s->flags(*c);
414 }

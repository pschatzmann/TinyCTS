101 NimBLEClient* connect() {
102 NimBLEScan* scan = NimBLEDevice::getScan();
103 scan->setActiveScan(true);
104 NimBLEScanResults results = scan->getResults(_scanMs, false);
105 const NimBLEAdvertisedDevice* found = nullptr;
106 for (int i = 0; i < results.getCount() && found == nullptr; i++) {
107 const NimBLEAdvertisedDevice* d = results.getDevice(i);
108 if (_device.empty()) {
109 if (d->isAdvertisingService(NimBLEUUID(CTS_SERVICE_UUID))) found = d;
110 } else if (isWanted(d->getAddress().toString(), d->getName())) {
111 found = d;
112 }
113 }
114 if (found == nullptr) return nullptr;
115 NimBLEClient* client = NimBLEDevice::createClient();
116 if (client == nullptr) return nullptr;
117 if (!client->connect(found)) {

119 return nullptr;

121 _name = found->getName();

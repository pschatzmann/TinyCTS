73 _roundTripMs = static_cast<uint32_t>(TINY_CTS_MILLIS() - t0);
74 ok = storeCurrentTime(data.data(), data.size(), t0, t0 + _roundTripMs);
75 if (!ok) _error = "the device does not know the time";

90 bool setTimeMs(uint64_t unixMs) override {
91 uint32_t start = TINY_CTS_MILLIS();
92 if (!connect()) return false;
93 // the round trip time of a read: the write needs about half of it
94 std::vector<uint8_t> data;
95 std::string ignored;
96 uint32_t t0 = TINY_CTS_MILLIS();
97 _central->read(CTS_CURRENT_TIME_UUID, data, ignored);
98 _roundTripMs = static_cast<uint32_t>(TINY_CTS_MILLIS() - t0);
99 uint8_t out[TinyCTSCurrentTime::Size];
100 size_t len = timeToWrite(
101 unixMs + static_cast<uint32_t>(TINY_CTS_MILLIS() - start) + _roundTripMs / 2, out);
102 bool ok = _central->write(CTS_CURRENT_TIME_UUID, out, len, _error);
103 if (!ok) _error = "rejected: " + _error;
104 _central->disconnect();
105 return ok;
106 }

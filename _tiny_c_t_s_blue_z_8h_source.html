50inline GVariant* call(const std::string& path, const char* iface, const char* method,
51 GVariant* params, std::string& error, int timeoutMs = 20000) {
52 GDBusConnection* conn = bus(error);
53 if (conn == nullptr) {
54 if (params != nullptr && g_variant_is_floating(params)) g_variant_unref(params);
55 return nullptr;
56 }
57 GError* err = nullptr;
58 GVariant* result = g_dbus_connection_call_sync(
59 conn, "org.bluez", path.c_str(), iface, method, params, nullptr,
60 G_DBUS_CALL_FLAGS_NONE, timeoutMs, nullptr, &err);
61 if (result == nullptr) {
62 g_dbus_error_strip_remote_error(err);
63 error = err->message;
64 if (verbose()) fprintf(stderr, "%s.%s failed: %s\n", iface, method, err->message);
65 g_error_free(err);
66 } else if (verbose()) {
67 fprintf(stderr, "%s.%s ok\n", iface, method);
68 }
69 return result;
70}

83inline std::string findAdapter(const std::string& name, std::string& error) {
84 GVariant* objs = call("/", "org.freedesktop.DBus.ObjectManager", "GetManagedObjects",
85 nullptr, error);
86 if (objs == nullptr) {
87 error = "cannot talk to BlueZ: " + error + " (is the bluetooth service running?)";
88 return "";
89 }
90 std::string wanted = "/org/bluez/" + name;
91 std::string found;
92 GVariantIter* it;
93 g_variant_get(objs, "(a{oa{sa{sv}}})", &it);
94 const char* path;
95 GVariant* ifaces;
96 while (g_variant_iter_next(it, "{&o@a{sa{sv}}}", &path, &ifaces)) {
97 GVariant* a = g_variant_lookup_value(ifaces, "org.bluez.Adapter1", nullptr);
98 if (a != nullptr) {
99 if (wanted == path) found = path;
100 g_variant_unref(a);
101 }
102 g_variant_unref(ifaces);
103 }
104 g_variant_iter_free(it);
105 g_variant_unref(objs);
106 if (found.empty()) error = "no Bluetooth adapter " + name + " found";
107 return found;
108}

110inline std::vector<Device> listDevices(const std::string& adapter) {
111 std::vector<Device> devices;
112 std::string error;
113 GVariant* objs = call("/", "org.freedesktop.DBus.ObjectManager", "GetManagedObjects",
114 nullptr, error);
115 if (objs == nullptr) return devices;
116 std::string cts = fullUuid(CTS_SERVICE_UUID);
117 GVariantIter* it;
118 g_variant_get(objs, "(a{oa{sa{sv}}})", &it);
119 const char* path;
120 GVariant* ifaces;
121 while (g_variant_iter_next(it, "{&o@a{sa{sv}}}", &path, &ifaces)) {
122 GVariant* p = g_variant_lookup_value(ifaces, "org.bluez.Device1", G_VARIANT_TYPE("a{sv}"));
123 if (p != nullptr && g_str_has_prefix(path, (adapter + "/").c_str())) {
124 Device d;
125 d.path = path;
126 const char* s;
127 if (g_variant_lookup(p, "Address", "&s", &s)) d.address = s;
128 if (g_variant_lookup(p, "Alias", "&s", &s)) d.name = s;
129 if (g_variant_lookup(p, "Name", "&s", &s)) d.name = s;
130 gint16 rssi;
131 if (g_variant_lookup(p, "RSSI", "n", &rssi)) {
132 d.seen = true;
133 d.rssi = rssi;
134 }
135 gboolean b;
136 if (g_variant_lookup(p, "ServicesResolved", "b", &b)) d.resolved = b;
137 if (g_variant_lookup(p, "Connected", "b", &b)) d.connected = b;
138 GVariant* uuids = g_variant_lookup_value(p, "UUIDs", G_VARIANT_TYPE("as"));
139 if (uuids != nullptr) {
140 gsize n;
141 const gchar** list = g_variant_get_strv(uuids, &n);
142 for (gsize i = 0; i < n; i++) {
143 if (cts == list[i]) d.cts = true;
144 }
145 g_free(list);
146 g_variant_unref(uuids);
147 }
148 devices.push_back(d);
149 }
150 if (p != nullptr) g_variant_unref(p);
151 g_variant_unref(ifaces);
152 }
153 g_variant_iter_free(it);
154 g_variant_unref(objs);
155 return devices;
156}

159inline std::string findCharacteristic(const std::string& devicePath, uint16_t uuid16) {
160 std::string result, error;
161 GVariant* objs = call("/", "org.freedesktop.DBus.ObjectManager", "GetManagedObjects",
162 nullptr, error);
163 if (objs == nullptr) return result;
164 std::string uuid = fullUuid(uuid16);
165 GVariantIter* it;
166 g_variant_get(objs, "(a{oa{sa{sv}}})", &it);
167 const char* path;
168 GVariant* ifaces;
169 while (g_variant_iter_next(it, "{&o@a{sa{sv}}}", &path, &ifaces)) {
170 GVariant* p = g_variant_lookup_value(ifaces, "org.bluez.GattCharacteristic1",
171 G_VARIANT_TYPE("a{sv}"));
172 if (p != nullptr) {
173 const char* s;
174 if (g_str_has_prefix(path, (devicePath + "/").c_str()) &&
175 g_variant_lookup(p, "UUID", "&s", &s) && uuid == s) {
176 result = path;
177 }
178 g_variant_unref(p);
179 }
180 g_variant_unref(ifaces);
181 }
182 g_variant_iter_free(it);
183 g_variant_unref(objs);
184 return result;
185}

187inline bool readValue(const std::string& characteristic, std::vector<uint8_t>& out,
188 std::string& error) {
189 GVariant* r = call(characteristic, "org.bluez.GattCharacteristic1", "ReadValue",
190 g_variant_new("(a{sv})", nullptr), error);
191 if (r == nullptr) return false;
192 GVariant* bytes = g_variant_get_child_value(r, 0);
193 gsize n;
194 const uint8_t* data = static_cast<const uint8_t*>(g_variant_get_fixed_array(bytes, &n, 1));
195 out.assign(data, data + n);
196 g_variant_unref(bytes);
197 g_variant_unref(r);
198 return true;
199}

218template <typename Found>
219inline std::vector<Device> discover(const std::string& adapter, uint32_t timeoutMs,
220 Found found, std::string& error) {
221 call(adapter, "org.bluez.Adapter1", "SetDiscoveryFilter",
222 g_variant_new_parsed("({'Transport': <'le'>, 'DuplicateData': <false>},)"), error);
223 bool started =
224 call(adapter, "org.bluez.Adapter1", "StartDiscovery", nullptr, error) != nullptr;
225 if (!started && error.find("InProgress") != std::string::npos) error.clear();
226 std::vector<Device> devices;
227 auto t0 = std::chrono::steady_clock::now();
228 for (;;) {
229 devices = listDevices(adapter);
230 if (found(devices)) break;
231 auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(
232 std::chrono::steady_clock::now() - t0)
233 .count();
234 if (ms >= timeoutMs) break;
235 g_usleep(250 * 1000);
236 }
237 std::string ignored;
238 if (started) call(adapter, "org.bluez.Adapter1", "StopDiscovery", nullptr, ignored);
239 return devices;
240}

26inline int32_t ctsDaysFromCivil(int32_t y, unsigned m, unsigned d) {
27 y -= m <= 2;
28 const int32_t era = (y >= 0 ? y : y - 399) / 400;
29 const unsigned yoe = static_cast<unsigned>(y - era * 400);
30 const unsigned doy = (153 * (m + (m > 2 ? -3 : 9)) + 2) / 5 + d - 1;
31 const unsigned doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
32 return era * 146097 + static_cast<int32_t>(doe) - 719468;
33}

36inline void ctsCivilFromDays(int32_t z, int32_t& y, unsigned& m, unsigned& d) {
37 z += 719468;
38 const int32_t era = (z >= 0 ? z : z - 146096) / 146097;
39 const unsigned doe = static_cast<unsigned>(z - era * 146097);
40 const unsigned yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;
41 const unsigned doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
42 const unsigned mp = (5 * doy + 2) / 153;
43 d = doy - (153 * mp + 2) / 5 + 1;
44 m = mp < 10 ? mp + 3 : mp - 9;
45 y = static_cast<int32_t>(yoe) + era * 400 + (m <= 2);
46}

90 static TinyCTSCurrentTime fromUnixMs(uint64_t unixMs, uint8_t reason = 0) {
91 TinyCTSCurrentTime t;
92 const uint32_t seconds = static_cast<uint32_t>(unixMs / 1000);
93 const int32_t days = static_cast<int32_t>(seconds / 86400);
94 const uint32_t rest = seconds % 86400;
95 int32_t y;
96 unsigned m, d;
97 ctsCivilFromDays(days, y, m, d);
98 t.year = static_cast<uint16_t>(y);
99 t.month = static_cast<uint8_t>(m);
100 t.day = static_cast<uint8_t>(d);
101 t.hours = static_cast<uint8_t>(rest / 3600);
102 t.minutes = static_cast<uint8_t>(rest % 3600 / 60);
103 t.seconds = static_cast<uint8_t>(rest % 60);
104 t.dayOfWeek = static_cast<uint8_t>((days + 3) % 7 + 1); // 1970-01-01: Thu
105 t.fractions256 = static_cast<uint8_t>(unixMs % 1000 * 256 / 1000);
106 t.adjustReason = reason;
107 return t;
108 }

115 bool toUnixMs(uint64_t& unixMs) const {
116 if (year < 1970 || year > 9999 || month < 1 || month > 12 || day < 1 ||
117 day > 31 || hours > 23 || minutes > 59 || seconds > 59) {
118 return false;
119 }
120 const int64_t days = ctsDaysFromCivil(year, month, day);
121 // reject days that a month does not have (31 February)
122 int32_t y;
123 unsigned m, d;
124 ctsCivilFromDays(static_cast<int32_t>(days), y, m, d);
125 if (y != year || m != month || d != day) return false;
126 unixMs = (static_cast<uint64_t>(days) * 86400 + hours * 3600UL +
127 minutes * 60UL + seconds) *
128 1000 +
129 fractions256 * 1000UL / 256;
130 return true;
131 }

135 out[0] = year & 0xFF;
136 out[1] = year >> 8;
137 out[2] = month;

153 static bool decode(const uint8_t* data, size_t len, TinyCTSCurrentTime& t) {
154 if (data == nullptr || len < Size - 1) return false;
155 t.year = static_cast<uint16_t>(data[0] | data[1] << 8);
156 t.month = data[2];

161 t.dayOfWeek = data[7];
162 t.fractions256 = data[8];
163 t.adjustReason = len >= Size ? data[9] : 0;
164 return true;

186 static TinyCTSLocalTimeInfo fromMinutes(int utcOffsetMinutes, int dstMinutes = 0) {
187 TinyCTSLocalTimeInfo i;
188 i.timeZone = static_cast<int8_t>(utcOffsetMinutes / 15);

53 bool begin(TinyCTSTimeSource& source, const char* deviceName = "TinyCTS") {
54 _source = &source;
55 if (!ctsArduinoBleBegin()) return false;
56 instance() = this;
57 if (_service == nullptr) {
58 uint16_t encryption = _settable && _encrypted ? BLEEncryption : 0;
59 uint16_t write = _settable ? BLEWrite : 0;
60 _service = new BLEService("1805");
61 _current = new BLECharacteristic("2A2B", BLERead | BLENotify | write | encryption,
62 TinyCTSCurrentTime::Size);
63 _local = new BLECharacteristic("2A0F", BLERead | write | encryption,
64 TinyCTSLocalTimeInfo::Size);
65 _reference = new BLECharacteristic("2A14", BLERead, TinyCTSReferenceTimeInfo::Size);
66 _current->setEventHandler(BLERead, onReadCurrent);
67 if (_settable) {
68 _current->setEventHandler(BLEWritten, onWriteCurrent);
69 _local->setEventHandler(BLEWritten, onWriteLocal);
70 }
71 _service->addCharacteristic(*_current);
72 _service->addCharacteristic(*_local);
73 _service->addCharacteristic(*_reference);
74 BLE.addService(*_service);
75 }
76 updateCurrent(0);
77 updateLocal();
78 updateReference();
79 BLE.setLocalName(deviceName);
80 BLE.setDeviceName(deviceName);
81 BLE.setAdvertisedService(*_service);
82 return BLE.advertise() != 0;
83 }

60 bool update() {
61 BLEClient* client = connect();
62 if (client == nullptr) return false;
63 bool ok = false;
64 BLERemoteService* service = client->service(BLEUUID(CTS_SERVICE_UUID));
65 if (service != nullptr) {
66 uint8_t data[16];
67 size_t len;
68 if (read(service->characteristic(BLEUUID(CTS_LOCAL_TIME_INFO_UUID)), data, len)) {
69 storeLocalInfo(data, len);
70 }
71 if (read(service->characteristic(BLEUUID(CTS_REFERENCE_TIME_INFO_UUID)), data, len)) {
72 storeReferenceInfo(data, len);
73 }
74 BLERemoteCharacteristic* chr = service->characteristic(BLEUUID(CTS_CURRENT_TIME_UUID));
75 uint32_t t0 = TINY_CTS_MILLIS();
76 if (read(chr, data, len)) {
77 ok = storeCurrentTime(data, len, t0, TINY_CTS_MILLIS());
78 }
79 }

105 BLEClient* connect() {
106 uint32_t seconds = (_scanMs + 999) / 1000;
107 if (seconds == 0) seconds = 1;
108 // the scan with the UUID keeps the devices that advertise the service
109 BLEScanReport* report =
110 _device.empty() ? BLE.scan(BLEUUID(CTS_SERVICE_UUID), seconds) : BLE.scan(seconds);
111 BLEClient* client = BLE.client();
112 bool connected = false;
113 for (auto& item : *report) {
114 std::string address = item.getAddress().toString().c_str();
115 std::string name = item.getName();
116 if (!_device.empty() && !isWanted(address, name)) continue;
117 connected = client->connect(item, 10);
118 if (connected) {
119 _address = address;
120 _name = name;
121 }
122 break;
123 }
124 BLE.clearScan();
125 return connected ? client : nullptr;
126 }

130 static bool read(BLERemoteCharacteristic* chr, uint8_t* out, size_t& len) {
131 if (chr == nullptr || !chr->canRead()) return false;
132 chr->getUInt8();
133 const void* value = chr->valueData();
134 len = chr->valueLen();
135 if (value == nullptr || len == 0) return false;
136 if (len > 16) len = 16;
137 memcpy(out, value, len);
138 return true;
139 }
